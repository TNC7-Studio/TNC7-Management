        import { initializeApp } from "https://www.gstatic.com/firebasejs/11.6.1/firebase-app.js";
        import { getAuth, signInAnonymously, signInWithCustomToken } from "https://www.gstatic.com/firebasejs/11.6.1/firebase-auth.js";
        import { getFirestore, doc, setDoc, deleteDoc, onSnapshot, collection, enableIndexedDbPersistence } from "https://www.gstatic.com/firebasejs/11.6.1/firebase-firestore.js";

        let state = { items: [], apiKeys: [], theme: 'dark', isLoggedIn: false };
        let db, auth;
        let deferredPrompt;

        const appId = typeof __app_id !== 'undefined' ? __app_id : 'default-app-id';
        const firebaseConfig = typeof __firebase_config !== 'undefined' ? JSON.parse(__firebase_config) : null;
        const el = (id) => document.getElementById(id);

        async function initApp() {
            setupPWA();
            const savedTheme = localStorage.getItem('TNC7_THEME');
            window.setTheme(savedTheme || 'dark');
            handleLoginSuccess();
            setInterval(() => { el('clock').innerText = new Date().toLocaleString('id-ID'); }, 1000);
            setupEventListeners();

            if (firebaseConfig) {
                try {
                    const app = initializeApp(firebaseConfig);
                    db = getFirestore(app);
                    enableIndexedDbPersistence(db).catch((err) => { console.warn("Offline persistence:", err.code); });
                    auth = getAuth(app);
                    
                    if (typeof __initial_auth_token !== 'undefined') await signInWithCustomToken(auth, __initial_auth_token);
                    else await signInAnonymously(auth);
                    
                    el('cloud-status-indicator').classList.remove('offline');
                    el('cloud-status-indicator').innerHTML = '<i class="fas fa-cloud-check"></i> Cloud Terhubung';
                    setupRealtimeListeners();
                } catch (e) {
                    setCloudOffline();
                }
            } else { setCloudOffline(); }
        }

        function setCloudOffline() {
            el('cloud-status-indicator').classList.add('offline');
            el('cloud-status-indicator').innerHTML = '<i class="fas fa-cloud-times"></i> Koneksi Cloud Terputus';
            showToast('Berjalan offline/lokal.');
        }

        function setupRealtimeListeners() {
            if (!db) return;
            const channelsRef = collection(db, 'artifacts', appId, 'public', 'data', 'channels');
            onSnapshot(channelsRef, (snapshot) => {
                state.items = [];
                snapshot.forEach((doc) => { state.items.push({ id: doc.id, ...doc.data() }); });
                state.items.sort((a,b) => (b.createdAt || 0) - (a.createdAt || 0));
                updateFilters(); renderTable();
            }, (err) => { setCloudOffline(); });

            const keysRef = collection(db, 'artifacts', appId, 'public', 'data', 'apikeys');
            onSnapshot(keysRef, (snapshot) => {
                state.apiKeys = [];
                snapshot.forEach((doc) => { state.apiKeys.push({ id: doc.id, ...doc.data() }); });
                renderApiKeys();
            });
        }

        function showToast(msg) {
            el('toast-msg').innerText = msg;
            el('toast').classList.add('show');
            setTimeout(() => el('toast').classList.remove('show'), 2500);
        }

        function handleLoginSuccess() {
            state.isLoggedIn = true; el('login-view').classList.add('hidden');
            el('dashboard-view').classList.remove('hidden'); updateFilters(); renderTable();
        }
        function handleLogout() {
            state.isLoggedIn = false; el('dashboard-view').classList.add('hidden');
            el('login-view').classList.remove('hidden');
        }

        window.setTheme = function(theme) {
            state.theme = theme; document.body.setAttribute('data-theme', theme);
            localStorage.setItem('TNC7_THEME', theme);
        };

            });
            
            el('item-form').addEventListener('submit', async (e) => {
                e.preventDefault();
                if(!db) { alert("Tidak ada koneksi Cloud!"); return; }
                const id = el('item-id').value; const docId = id || Date.now().toString();
                const newItem = {
                    id: docId, title: el('item-title').value, channelId: el('item-channelId').value,
                    link: el('item-link').value, pengelola: el('item-pengelola').value.toUpperCase(),
                    subs: el('item-subs').value || '0', subsGrowth: parseInt(el('item-subsGrowth').value) || 0,

        function updateFilters() {
            const select = el('filter-pengelola');
            const unique = [...new Set(state.items.map(i => i.pengelola).filter(Boolean))];
            select.innerHTML = '<option value="all">SEMUA PENGELOLA</option>' + unique.map(p => `<option value="${p}">${p}</option>`).join('');

        function renderTable() {
            const tbody = el('table-body');
            const sq = el('search-input').value.toLowerCase();
            const fm = el('filter-monet').value;
            const fp = el('filter-pengelola').value;
            const sort = el('sort-data').value;

            let filtered = state.items.filter(i => {
                const ms = i.title.toLowerCase().includes(sq) || (i.niche && i.niche.toLowerCase().includes(sq)) || (i.channelId && i.channelId.toLowerCase().includes(sq));
                const mm = fm === 'all' ? true : (fm === 'yes' ? i.monet === 'Sudah' : i.monet !== 'Sudah');
                const mp = fp === 'all' ? true : i.pengelola === fp;
                return ms && mm && mp;
            });

            if(sort === 'views_desc') filtered.sort((a,b) => (b.viewGrowth||0) - (a.viewGrowth||0));
            if(sort === 'subs_desc') filtered.sort((a,b) => (b.subsGrowth||0) - (a.subsGrowth||0));

            const now = Date.now();
            const newChannelsCount = state.items.filter(i => (now - (i.createdAt || 0)) <= 86400000).length;

            el('stat-total').innerText = state.items.length;

                        <td><div class="font-bold text-sm mb-1">${item.title}</div><div>${badge}</div></td>

                        <td class="font-bold text-sm">${item.views||'0'}</td><td>${vG}</td>
                        <td class="text-center font-semibold text-sm">${item.videoCount}</td>

        window.addApiKey = async function() {
            if(!db) { alert('Tidak ada koneksi Cloud!'); return; }
            const keyVal = el('new-api-key').value.trim(); let labelVal = el('new-api-label').value.trim();
            if(!keyVal) { showToast('API Key kosong!'); return; }
            if(!labelVal) labelVal = 'Key ' + ((state.apiKeys?.length || 0) + 1);
            
            try {
                const docId = Date.now().toString();
                await setDoc(doc(db, 'artifacts', appId, 'public', 'data', 'apikeys', docId), { key: keyVal, label: labelVal, active: state.apiKeys.length === 0, createdAt: Date.now() });
                el('new-api-key').value = ''; showToast('API Key tersimpan ke Cloud!');
            } catch(e) { alert("Gagal menyimpan API Key."); }
        };

            }
        };

        window.searchYouTubeChannel = async function() {
            const query = el('yt-search-input').value.trim();
            if (!query) return;

            const activeKeyObj = state.apiKeys.find(k => k.active);
            if (!activeKeyObj || !activeKeyObj.key) {
                alert('API Key YouTube belum aktif!'); window.closeModal('modal-search-yt'); window.openModal('modal-settings'); return;
            }
            const apiKey = activeKeyObj.key;
            let handle = query.includes('@') ? query.split('@')[1].split('/')[0].split('?')[0] : null;

            const btn = el('btn-yt-search'); const originalText = btn.innerHTML;
            btn.innerHTML = '<i class="fas fa-spinner fa-spin"></i> Mencari...'; btn.disabled = true;

            try {
                let channelData = null;
                if (handle) {
                    const res = await fetch(`https://www.googleapis.com/youtube/v3/channels?part=snippet,statistics&forHandle=${handle}&key=${apiKey}`);
                    const data = await res.json(); if (data.items) channelData = data.items[0];
                } else {
                    const searchRes = await fetch(`https://www.googleapis.com/youtube/v3/search?part=snippet&type=channel&q=${encodeURIComponent(query)}&key=${apiKey}`);
                    const searchData = await searchRes.json();
                    if (searchData.items && searchData.items.length > 0) {
                        const statRes = await fetch(`https://www.googleapis.com/youtube/v3/channels?part=snippet,statistics&id=${searchData.items[0].snippet.channelId}&key=${apiKey}`);
                        const statData = await statRes.json(); if (statData.items) channelData = statData.items[0];
                    }
                }

                if (channelData) {
                    const formatNumber = (numStr) => {
                        const num = parseInt(numStr);
                        if (isNaN(num)) return '0';
                        if (num >= 1000000) return (num / 1000000).toFixed(1).replace('.0', '') + 'M';
                        if (num >= 1000) return (num / 1000).toFixed(1).replace('.0', '') + 'K';
                        return num.toString();
                    };

                    window.closeModal('modal-search-yt'); window.editItem(null); 
                    el('item-title').value = channelData.snippet.title; el('item-channelId').value = channelData.id;
                    el('item-link').value = channelData.snippet.customUrl ? `https://youtube.com/${channelData.snippet.customUrl}` : `https://youtube.com/channel/${channelData.id}`;
                    el('item-subs').value = formatNumber(channelData.statistics.subscriberCount);
                    el('item-views').value = formatNumber(channelData.statistics.viewCount);
                    el('item-video').value = channelData.statistics.videoCount;
                    el('yt-search-input').value = ''; showToast('Data ditemukan!');
                } else { alert('Channel tidak ditemukan.'); }
            } catch (err) { alert('Terjadi kesalahan pencarian API.'); } 
            finally { btn.innerHTML = originalText; btn.disabled = false; }
        };

        function setupPWA() {
            const manifest = {
                "name": "TNC7 Management", "short_name": "TNC7",
                "start_url": window.location.pathname, "display": "standalone",
                "background_color": "#161921", "theme_color": "#161921",
                "icons": [{ "src": "https://upload.wikimedia.org/wikipedia/commons/e/ef/Youtube_logo.png", "sizes": "512x512", "type": "image/png", "purpose": "any maskable" }]
            };
            
            const manifestUrl = 'data:application/manifest+json;charset=utf-8,' + encodeURIComponent(JSON.stringify(manifest));
            let link = document.querySelector('link[rel="manifest"]');
            if (!link) { link = document.createElement('link'); link.rel = 'manifest'; document.head.appendChild(link); }
            link.href = manifestUrl;

            if ('serviceWorker' in navigator) {
                const swCode = `
                    self.addEventListener('install', (e) => self.skipWaiting());
                    self.addEventListener('activate', (e) => self.clients.claim());
                    self.addEventListener('fetch', (e) => {
                        e.respondWith(fetch(e.request).catch(err => console.log('Bypass SW', err)));
                    });
                `;
                const swUrl = URL.createObjectURL(new Blob([swCode], { type: 'application/javascript' }));
                navigator.serviceWorker.register(swUrl).then(() => {
                    console.log("PWA Service Worker registered.");
                }).catch((err) => { console.warn("SW gagal:", err); });
            }

            window.addEventListener('beforeinstallprompt', (e) => {
                e.preventDefault(); deferredPrompt = e;
                const installBtn = document.getElementById('btn-install');
                if(installBtn) { installBtn.classList.remove('hidden'); installBtn.style.animation = 'pulse 2s infinite'; }
            });

            const installBtn = document.getElementById('btn-install');
            if(installBtn) {
                installBtn.addEventListener('click', async () => {
                    if (deferredPrompt) {
                        deferredPrompt.prompt();
                        const { outcome } = await deferredPrompt.userChoice;
                        if (outcome === 'accepted') installBtn.classList.add('hidden');
                        deferredPrompt = null;
                    }
                });
            }
        }
